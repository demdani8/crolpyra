using System;
using System.Configuration;

using System.Data.SqlClient;

using System.Web.UI;


public partial class InvProjects : System.Web.UI.Page
{
    

    protected void Page_Load(object sender, EventArgs e)
    {
        if (!IsPostBack) Bind(); LoadKpis2();
    }

    private void Bind(string filter = null)
    {
        string sql = "SELECT ProjectId, Name, Code, [User], IsActive, Place, SecUser FROM dbo.DLI_Projects";
        if (!string.IsNullOrEmpty(filter))
        {
            sql += " WHERE Name LIKE @f OR Code LIKE @f";
        }
        sql += " ORDER BY Name";

        using (var cn = new SqlConnection(ConnStr))
        using (var cmd = new SqlCommand(sql, cn))
        {
            if (!string.IsNullOrEmpty(filter))
                cmd.Parameters.AddWithValue("@f", "%" + filter + "%");

            using (var da = new SqlDataAdapter(cmd))
            {
                var dt = new DataTable();
                da.Fill(dt);
                gv.DataSource = dt;
                gv.DataBind();
            }
        }
    }
    protected void btnSearch_Click(object sender, EventArgs e)
    {
        string f = txtSearch.Text.Trim();
        Bind(f);
    }

    protected void btnClear_Click(object sender, EventArgs e)
    {
        txtSearch.Text = "";
        Bind();
    }

    protected void btnAdd_Click(object sender, EventArgs e)
    {
        string name = (txtNewName.Text ?? "").Trim();
        string code = (txtNewCode.Text ?? "").Trim();
        string user = (txtNewUser.Text ?? "").Trim();
        string secuser = (txtNewUserSecond.Text ?? "").Trim();
        string place = (txtNewPlace.Text ?? "").Trim();
        if (name.Length == 0) { lblMsg.Text = "Adj nevet."; return; }
        using (var cn = new SqlConnection(ConnStr))
        using (var cmd = new SqlCommand("INSERT INTO dbo.DLI_Projects(Name, Code, IsActive, CreatedAt, [User], Place, SecUser) VALUES(@n,@c,1,SYSDATETIME(), @u, @place, @secuser)", cn))
        {
            cmd.Parameters.AddWithValue("@n", name);
            cmd.Parameters.AddWithValue("@u", user);
            cmd.Parameters.AddWithValue("@place", place);
            cmd.Parameters.AddWithValue("@secuser", secuser);
            cmd.Parameters.AddWithValue("@c", (object)(code.Length == 0 ? null : code) ?? DBNull.Value);
            cn.Open(); cmd.ExecuteNonQuery();
        }
        txtNewName.Text = ""; txtNewCode.Text = "";  txtNewUser.Text = ""; lblMsg.Text = "Hozzáadva.";
        Bind();
    }

    protected void gv_RowEditing(object sender, GridViewEditEventArgs e)
    {
        gv.EditIndex = e.NewEditIndex; Bind();
    }

    protected void gv_RowCancelingEdit(object sender, GridViewCancelEditEventArgs e)
    {
        gv.EditIndex = -1; Bind();
    }

    protected void gv_RowUpdating(object sender, GridViewUpdateEventArgs e)
    {
        int id = (int)gv.DataKeys[e.RowIndex].Value;
        GridViewRow row = gv.Rows[e.RowIndex];
        string name = ((TextBox)row.Cells[0].Controls[0]).Text.Trim();
        string code = ((TextBox)row.Cells[1].Controls[0]).Text.Trim();
        string user = ((DropDownList)row.FindControl("ddlUser")).SelectedValue;
        string secuser = ((TextBox)row.Cells[4].Controls[0]).Text.Trim();
        string place = ((TextBox)row.Cells[3].Controls[0]).Text.Trim();
        bool isActive = ((CheckBox)row.Cells[5].Controls[0]).Checked;

        if (name.Length == 0) { lblMsg.Text = "Név kötelező."; return; }

        using (var cn = new SqlConnection(ConnStr))
        using (var cmd = new SqlCommand("UPDATE dbo.DLI_Projects SET Name=@n, Code=@c, IsActive=@a, [User]=@u, Place=@place, SecUser=@secuser WHERE ProjectId=@id", cn))
        {
            cmd.Parameters.AddWithValue("@n", name);
            cmd.Parameters.AddWithValue("@u", user);
            cmd.Parameters.AddWithValue("@place", place);
            cmd.Parameters.AddWithValue("@secuser", secuser);
            cmd.Parameters.AddWithValue("@c", (object)(code.Length == 0 ? null : code) ?? DBNull.Value);
            cmd.Parameters.AddWithValue("@a", isActive ? 1 : 0);
            cmd.Parameters.AddWithValue("@id", id);
            cn.Open(); cmd.ExecuteNonQuery();
        }
        gv.EditIndex = -1; lblMsg.Text = "Mentve."; Bind();
    }

     protected void gv_RowDeleting(object sender, GridViewDeleteEventArgs e)
    {
        int id = (int)gv.DataKeys[e.RowIndex].Value;
        GridViewRow row = gv.Rows[e.RowIndex];

        using (var cn = new SqlConnection(ConnStr))
        {
            cn.Open();
            using (var cmd = new SqlCommand("Delete from DLI_PartAssignments WHERE ProjectId=@id", cn))
            {
                
                cmd.Parameters.AddWithValue("@id", id);
                 cmd.ExecuteNonQuery();
            }
            using (var cmd2 = new SqlCommand("Delete from DLI_Projects WHERE ProjectId=@id", cn))
            {
                
                cmd2.Parameters.AddWithValue("@id", id);
                 cmd2.ExecuteNonQuery();
            }
        }
        gv.EditIndex = -1; lblMsg.Text = "Törölve."; Bind();
    }


     private void LoadKpis2()
    {
        using (var cn = new SqlConnection(ConnStr))
        using (var cmd = new SqlCommand(@"
            SELECT
              (SELECT COUNT(*) FROM DLI_Projects) AS AllCnt,
              (SELECT COUNT(*) FROM DLI_Projects WHERE IsActive=1) AS AssignedCnt,
              (SELECT COUNT(*) FROM DLI_Projects WHERE IsActive=0) AS FreeCnt;", cn))
        {
            cn.Open();
            using (var rd = cmd.ExecuteReader())
            {
                if (rd.Read())
                {
                    pAll.Text = rd["AllCnt"].ToString();
                    pActive.Text = rd["AssignedCnt"].ToString();
                    pInactive.Text = rd["FreeCnt"].ToString();
                }
            }
        }
    }

    protected void gv_RowDataBound(object sender, GridViewRowEventArgs e)
    {
        if (e.Row.RowType == DataControlRowType.DataRow && e.Row.RowState != DataControlRowState.Edit)
        {
            string projectId = gv.DataKeys[e.Row.RowIndex].Value.ToString();

            // kattintás az egész sorra -> részletek oldal
            e.Row.Attributes["onclick"] = "window.location='InvDetails.aspx?id=" + projectId + "';";

            // mutató változzon kézre
            e.Row.Style["cursor"] = "pointer";

            // DE: a művelet oszlopban lévő gombokra ne hasson
            foreach (TableCell cell in e.Row.Cells)
            {
                foreach (Control ctrl in cell.Controls)
                {
                    if (ctrl is IButtonControl || ctrl is HyperLink)
                    {
                        // ha gomb van, állítsd le a "sorkattintást"
                        if (ctrl is WebControl)
                        {
                            WebControl wc = (WebControl)ctrl;
                            wc.Attributes["onclick"] = "event.stopPropagation();";
                        }
                    }
                }
            }
        }
    }
}
